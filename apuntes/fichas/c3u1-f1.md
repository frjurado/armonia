arriba = {
  \modelo \encima "G" \analitico <g' b' d''>1 \finModelo
  \encima "G/B"  \analitico <g' b' d''>1
  \encima "C"    \analitico <c'' e'' g''>1
  \encima "F♯°/A" \analitico <fs' a' c''>1
  \encima "G/D"  \analitico <g' b' d''>1
  \encima "Em"   \analitico <e' g' b'>1
  \encima "D/F♯" \analitico <d'' fs'' a''>1
  \encima "Bm"   \analitico <b' d'' fs''>1
}
abajo = {
  \modelo         \gradoBajo "①" \acorde "I" \finModelo g,1
  \figuras "6"    \gradoBajo "③" \acorde "I6"    b,1
                  \gradoBajo "④" \acorde "IV"    c1
  \figuras "6"    \gradoBajo "②" \acorde "VII6"  a,1
  \figuras "6/4"  \gradoBajo "⑤" \acorde "I6/4"  d1
                  \gradoBajo "⑥" \acorde "VI"    e1
  \figuras "6"    \gradoBajo "⑦" \acorde "V6"    fs,1
                  \gradoBajo "③" \acorde "III"   b,1
}

::: {.ejercicio tipo="analisis" tono="Sol mayor" obra="R. Schumann, Álbum de la juventud, op. 68, n.º 2, «Marcha del soldado», cc. 1–4"}
```lilypond
%% Notas de la edición Peters, según la transcripción de Philippe
%% Hézaine para el Mutopia Project (CC BY-SA 2.5), pasadas a alturas
%% absolutas. La semicorchea del c. 1 (la en el bajo) es de paso entre
%% I e I6: sin cifrar.
arriba = {
  \time 2/4
  \tempo "Munter und straff"
  <g' b'>8.\f <g' c''>16 <g' d''>8 r |
  <g' e''>8 r <g' d''> r |
  <fs' c''> r <g' b'> r |
  <fs' a'> r g' r |
}
abajo = {
  \time 2/4
  \modelo \acorde "I" \finModelo g8. a16 \acorde "I6" b8 r |
  \acorde "IV" c'8 r \acorde "I6" b r |
  \acorde "VII6" a r \acorde "I" g r |
  \acorde "V7" <d c'> r \acorde "I" <g b> r |
}
```
